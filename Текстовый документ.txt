    <!-- Modal -->
    <div class="modal" id="carModal">
        <div class="modal-content">
            <span class="close" onclick="closeModal()">&times;</span>
            <h3>Նոր մեքենա</h3>
            <form onsubmit="addCar(event)">
                <label>Մակնիշ և Մոդել</label>
                <input type="text" id="title" placeholder="Օր. Toyota Camry" required>
                
                <label>Գին ($)</label>
                <input type="number" id="price" placeholder="Օր. 12000" required>
                
                <label>Տարեթիվ</label>
                <input type="number" id="year" placeholder="Օր. 2015" required>
                
                <label>Վազք</label>
                <input type="text" id="mileage" placeholder="Օր. 140,000 կմ" required>
                
                <label>Գտնվելու վայրը</label>
                <input type="text" id="location" placeholder="Օր. Երևան" required>

                <label>Ընտրել նկար պատկերասրահից</label>
                <input type="file" id="carImageFile" accept="image/*" required>
                
                <label>Հեռախոսահամար</label>
                <input type="text" id="phone" placeholder="Օր. +374 91 123456" required>
                
                <button type="submit" class="btn" style="width: 100%; margin-top: 10px;">Տեղադրել</button>
            </form>

    <script>
        // Նախնական կամ պահպանված տվյալներ
        let cars = JSON.parse(localStorage.getItem('am_cars')) || [
            {
                title: "Toyota Camry",
                price: 14500,
                year: 2016,
                mileage: "120,000 կմ",
                location: "Երևան",
                image: "https://images.unsplash.com/photo-1621007947382-bb3c3994e3fb?auto=format&fit=crop&w=600&q=80",
                phone: "+374 91 123456"
            }
        ];

        function renderCars(data) {
            const grid = document.getElementById('carGrid');
            grid.innerHTML = '';
            
            if (data.length === 0) {
                grid.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: #666;">Հայտարարություններ չկան</p>';
                return;
            }

            data.forEach(car => {
                grid.innerHTML += `
                    <div class="card">
                        <img src="${car.image}" alt="${car.title}" onerror="this.src='https://images.unsplash.com/photo-1533473359331-0135ef1b58bf?auto=format&fit=crop&w=600&q=80'">
                        <div class="card-body">
                            <div class="card-title">${car.title} (${car.year})</div>
                            <div class="card-price">$${car.price.toLocaleString()}</div>
                            <div class="card-details">
                                📍 ${car.location} | ⚙️ ${car.mileage}
                            </div>
                            <div class="card-phone">📞 ${car.phone}</div>
                        </div>
                    </div>
                `;
            });
        }

        function filterCars() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const filtered = cars.filter(c => c.title.toLowerCase().includes(query));
            renderCars(filtered);
        }

        function openModal() {
            document.getElementById('carModal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('carModal').style.display = 'none';
        }

        function addCar(e) {
            e.preventDefault();
            
            const fileInput = document.getElementById('carImageFile');
            const file = fileInput.files[0];

            if (file) {
                const reader = new FileReader();
                reader.onload = function(event) {
                    const newCar = {
                        title: document.getElementById('title').value,
                        price: Number(document.getElementById('price').value),
                        year: Number(document.getElementById('year').value),
                        mileage: document.getElementById('mileage').value,
                        location: document.getElementById('location').value,
                        image: event.target.result,
                        phone: document.getElementById('phone').value
                    };
                    
                    cars.unshift(newCar);
                    localStorage.setItem('am_cars', JSON.stringify(cars));
                    renderCars(cars);
                    closeModal();
                    e.target.reset();
                };
                reader.readAsDataURL(file);
            }
        }

        // Սկզբնական բեռնում
        renderCars(cars);
    </script>
